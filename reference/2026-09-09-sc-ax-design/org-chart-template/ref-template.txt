class Component extends DCLogic {
  state = { kind: 'primary', grantStd: true, revokePrev: false, reason: '', confirmOpen: false };

  renderVals() {
    const s = this.state;
    const isPrimary = s.kind === 'primary';
    return {
      revokedHistory: this.props.revokedHistory ?? true,
      changeLogEmpty: this.props.changeLogEmpty ?? false,
      changeLogFilled: !(this.props.changeLogEmpty ?? false),

      isPrimary,
      isConcurrent: !isPrimary,
      pickPrimary: () => this.setState({ kind: 'primary' }),
      pickConcurrent: () => this.setState({ kind: 'concurrent', revokePrev: false }),

      grantStd: s.grantStd,
      toggleGrantStd: (v) => this.setState({ grantStd: v }),
      revokePrev: s.revokePrev && isPrimary,
      revokeDisabled: !isPrimary,
      toggleRevokePrev: (v) => this.setState({ revokePrev: v }),

      reason: s.reason,
      setReason: (v) => this.setState({ reason: v }),
      saveBlocked: (this.props.reasonRequired ?? true) && !s.reason.trim(),

      confirmOpen: s.confirmOpen,
      openConfirm: () => this.setState({ confirmOpen: true }),
      closeConfirm: () => this.setState({ confirmOpen: false }),
    };
  }
}
